-- 텔레포트 함수
local function LookTeleport()
    local cam = workspace.CurrentCamera
    local char = game.Players.LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    
    if hrp and cam then
        local rayOrigin = cam.CFrame.Position
        local rayDirection = cam.CFrame.LookVector * 1000
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
        raycastParams.FilterDescendantsInstances = {char}
        
        local raycastResult = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
        
        local targetPos
        if raycastResult then
            targetPos = raycastResult.Position + Vector3.new(0, 3, 0)
        else
            targetPos = rayOrigin + (rayDirection * 0.5)
        end
        
        hrp.CFrame = CFrame.new(targetPos)
    end
end

    local dragStart
    local startPos

    button.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = button.Position
        end

        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

local function TPtoTarget(target)
    local TCHAR = target.Character
    if not TCHAR then return false end
    local THRP = TCHAR:FindFirstChild("Torso") or TCHAR:FindFirstChild("HumanoidRootPart")
    if not THRP then return false end

    local localChar = plr.Character
    local localHRP = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not localHRP then return false end

    if TCHAR and THRP and localHRP then
        local ping = plr:GetNetworkPing()
        local offset = THRP.Position + (THRP.Velocity * (ping + 0.15))
        localHRP.CFrame = CFrame.new(offset) * THRP.CFrame.Rotation
        return true
    end
    return false
end

        hrp.AssemblyAngularVelocity = Vector3.zero
    end
end

-- [ 오너킥 (Y 9999999로 날리기) ]
-- =============================================
local function doKick(player)
    if not player or not player.Character then return false end
    
    local char = player.Character
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local torso = char:FindFirstChild("Torso")
    local hum = char:FindFirstChild("Humanoid")
    
    if not hrp then return false end
    
    pcall(function()
        if SetNetworkOwner then
            SetNetworkOwner:FireServer(hrp, hrp.CFrame)
            if torso then SetNetworkOwner:FireServer(torso, torso.CFrame) end
        end
    end)
    task.wait(0.05)
    
    local targetPos = Vector3.new(hrp.Position.X, KICK_Y, hrp.Position.Z)
    hrp.CFrame = CFrame.new(targetPos)
    hrp.AssemblyLinearVelocity = Vector3.new(0, 50000, 0)
    
    if torso then
        torso.CFrame = CFrame.new(targetPos)
        torso.AssemblyLinearVelocity = Vector3.new(0, 50000, 0)
    end
    
    if DestroyGrabLine then
        pcall(function() DestroyGrabLine:FireServer(hrp) end)
    end
    
    if hum then
        hum.Sit = true
        hum.PlatformStand = true
    end
    
    return true
end

    end

local function kickLoop()
    while isKickActive do
        if #targetList == 0 then
            task.wait(0.5)
            continue
        end
        
        for _, targetName in ipairs(targetList) do
            if not isKickActive then break end
            
            local player = Players:FindFirstChild(targetName)
            if player and player.Character then
                saveMyPosition()
                TPtoTarget(player)
                task.wait(0.1)
                doKick(player)
                task.wait(0.1)
                restoreMyPosition()
            end
            task.wait(0.2)
        end
        task.wait(0.3)
    end
end

-- 킥 알림 함수
local function setupKickNotifications()
    -- 플레이어 추가될 때
    Players.PlayerAdded:Connect(function(player)
        player.CharacterAdded:Connect(function(char)
            local hrp = char:WaitForChild("HumanoidRootPart", 1)
            if hrp then
                anchoredCache[player] = false
            end
        end)
    end)

    -- 플레이어 나갈 때 (킥 감지)
    Players.PlayerRemoving:Connect(function(player)
        if anchoredCache[player] == true and kickNotificationsEnabled then
            Rayfield:Notify({
                Title = "👢 Kick 감지",
                Content = string.format("%s (@%s) 님이 킥당했습니다", player.DisplayName, player.Name),
                Duration = 5
            })
        end
        anchoredCache[player] = nil
    end)

    -- 실시간 Anchored 상태 체크
    task.spawn(function()
        while kickNotificationsEnabled do
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= plr then
                    local char = player.Character
                    if char then
                        local hrp = char:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            anchoredCache[player] = hrp.Anchored == true
                        end
                    end
                end
            end
            task.wait()
        end
    end)
end

        local myChar = plr.Character
        if not myChar then 
            task.wait(0.5)
            continue 
        end
        
        local myHRP = myChar:FindFirstChild("HumanoidRootPart")

            local char = player.Character
            if not char then continue end
            
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            
            local hum = char:FindFirstChild("Humanoid")
            if not hum or hum.Health <= 0 then continue end -- 죽은 사람 제외
            
            -- 거리 계산
            local distance = (myHRP.Position - hrp.Position).Magnitude
            if distance <= AuraRange then
                -- 루프그랩 실행 (SetNetworkOwner만 사용)
                pcall(function()
                    if SetNetworkOwner then
                        SetNetworkOwner:FireServer(hrp, CFrame.lookAt(myHRP.Position, hrp.Position))
                    end
                end)
            end
        end
        
        task.wait(0.1) -- 0.1초마다 체크
    end
end

function TP(target)
    local TCHAR = target.Character
    if not TCHAR then return false end
    local THRP = TCHAR:FindFirstChild("Torso") or TCHAR:FindFirstChild("HumanoidRootPart")
    if not THRP then return false end

    local localChar = plr.Character
    local localHRP = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not localHRP then return false end

    local dist = (THRP.Position - localHRP.Position).Magnitude

    if TCHAR and THRP and localHRP then
        local ping = game:GetService("Players").LocalPlayer:GetNetworkPing()
        local offset = THRP.Position + (THRP.Velocity * (ping + 0.15))

        if dist > 30 then 
            localHRP.CFrame = CFrame.new(offset) * THRP.CFrame.Rotation * CFrame.new(0,-13,0)
            localHRP.AssemblyLinearVelocity = Vector3.zero
            localHRP.AssemblyAngularVelocity = Vector3.zero
        end

        return true
    end
    return false
end

function SetOwner(target)
    local head = target.Character:FindFirstChild("HumanoidRootPart")
    if head and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
        rs.GrabEvents.SetNetworkOwner:FireServer(head, head.CFrame)
        return true
    end
    return false
end

function UnOwner(target)
    local head = target.Character:FindFirstChild("HumanoidRootPart")
    if head and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
        rs.GrabEvents.DestroyGrabLine:FireServer(head, head.CFrame)
        return true
    end
    return false
end

function BACK(originCF)
    if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
        plr.Character.HumanoidRootPart.CFrame = originCF
    end
end

local function safeGetCharacterParts(player)

    return char, hrp, head
end

        end

    end

    local function saveMyOriginalPosition()

        local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if myHRP then
            myOriginalPosition = myHRP.CFrame
        end
    end

        local myChar = plr.Character
        local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if myHRP and myOriginalPosition then
            myHRP.CFrame = myOriginalPosition
        end
    end

        if not player then return end

        while blobLoopT4 do
            if not player.Parent then
                monitoredPlayers[player] = nil
                independentHoldConnections[player] = nil
                if anchorToggleTasks[player] then
                    task.cancel(anchorToggleTasks[player])

                end
                return

                if not player.Parent then return end
                task.wait(0.3)
                character = player.Character
            end

                task.wait(0.3)
                continue
            end

            if not myOriginalPosition then
                saveMyOriginalPosition()
            end

                if not player.Parent then return end
                if ownerTags[player] then
                    ownerTags[player]:Destroy()
                    ownerTags[player] = nil
                end

                if independentHoldConnections[player] then
                    independentHoldConnections[player]:Disconnect()
                    independentHoldConnections[player] = nil
                end

                if anchorToggleTasks[player] then
                    task.cancel(anchorToggleTasks[player])
                    anchorToggleTasks[player] = nil
                end

                cleanupBodyPosition(player)
                task.wait(0.1)
                character = player.Character
                charHUM = character and character:FindFirstChild("Humanoid")
            end

            local _, safeHRP, head = safeGetCharacterParts(player)
            if not safeHRP or not head then
                task.wait(0.3)
                continue
            end

            local myChar = plr.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myHRP then
                task.wait(0.3)
                continue
            end

            local charHRP = character:FindFirstChild("HumanoidRootPart")
            local charTOR = character:FindFirstChild("Torso")
            if not charHRP then
                task.wait(0.3)
                continue
            end

            if not ownerKickInitialized then
                if charHUM and myHRP and charHRP then
                    for i = 1, 30 do 
                        rs.GrabEvents.SetNetworkOwner:FireServer(charHRP, CFrame.lookAt(myHRP.Position, charHRP.Position))
                    end
                    task.wait(0.1)
                end
                ownerKickInitialized = true
            end

            -- 거리 계산
            local distance = (myHRP.Position - charHRP.Position).Magnitude
            local wasFar = distance > 30
            local originalPos = myHRP.CFrame

            -- 30스터드 이상 차이나면 상대방에게 TP
            if wasFar then
                TP(player)
                task.wait(0.2)
            end

            local tpRunning = true
            local CF
            task.spawn(function()
                while tpRunning and blobLoopT4 do
                    if not player.Parent then break end
                    local ok, cf = TP(player)
                    if ok and cf then 
                        CF = cf
                    end
                    task.wait()
                end
            end)

            local shouldContinue = false
            while blobLoopT4 do
                if not character.Parent or not charHUM or charHUM.Health <= 0 then 
                    tpRunning = false
                    shouldContinue = true

                    local myChar = plr.Character
                    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
                    if myHRP and originPositions[player] then
                        myHRP.CFrame = originPositions[player]
                    end
                    break
                end

                local mySeat = plr.Character and plr.Character:FindFirstChildOfClass("Humanoid") and plr.Character:FindFirstChildOfClass("Humanoid").SeatPart
                local amIRiding = mySeat and mySeat.Parent and mySeat.Parent.Name == "CreatureBlobman"

                if myOriginalPosition then
                    charHRP.CFrame = myOriginalPosition * CFrame.new(0, OLTPValue.Y, 0)

                end

                elseif originPositions[player] then
                    targetCFrame = originPositions[player]
                    BACK(originPositions[player])
                end

                    myHRP.CFrame = myOriginalPosition
                end

                if anchorToggleTasks[player] then
                    task.cancel(anchorToggleTasks[player])
                    anchorToggleTasks[player] = nil
                end

                            if table.find(targetNames, item.Name) and item:FindFirstChild("StickyPart") then

                                local weld = sticky:FindFirstChild("StickyWeld")
                                if weld and weld:IsA("WeldConstraint") and weld.Part1 and weld.Part1:IsDescendantOf(character) then

                                    sticky.CFrame = CFrame.new(0,9999,0)
                                end
                            end
                        end
                    end
                end

                if independentHoldConnections[player] then
                    independentHoldConnections[player]:Disconnect()
                    independentHoldConnections[player] = nil
                end

                cleanupBodyPosition(player)
                local bp = Instance.new("BodyPosition")
                bp.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                bp.P = 800000
                bp.D = 500
                bp.Parent = charHRP
                bodyPositions[player] = bp

                independentHoldConnections[player] = RunService.Heartbeat:Connect(function()
                    if not blobLoopT4 or not character or not character.Parent or not charHRP or not charHRP.Parent or not charHUM or charHUM.Health <= 0 then
                        if independentHoldConnections[player] then 
                            independentHoldConnections[player]:Disconnect()
                            independentHoldConnections[player] = nil
                            destroyCounts[player] = nil
                        end
                        if anchorToggleTasks[player] then
                            task.cancel(anchorToggleTasks[player])
                            anchorToggleTasks[player] = nil
                        end
                        if ownerTags[player] then
                            ownerTags[player]:Destroy()
                            ownerTags[player] = nil
                        end
                        if charTOR then
                            charTOR.Anchored = false

                        if not charHUM.Sit and not OnlyOwner then

                            rs.GrabEvents.DestroyGrabLine:FireServer(charHRP) 
                        end
                        if player.IsHeld then 
                            rs.GrabEvents.SetNetworkOwner:FireServer(charHRP, CFrame.lookAt(myHRP.Position, charHRP.Position)) 
                        end
                        if charHUM.Sit and not OnlyOwner then 
                            rs.GrabEvents.DestroyGrabLine:FireServer(charHRP) 
                        end
                    end
                    
                    if not isRidingNow and OwnerKickMODED then
                        rs.GrabEvents.SetNetworkOwner:FireServer(charHRP, CFrame.lookAt(myHRP.Position, charHRP.Position))
                        if not OnlyOwner then 
                            rs.GrabEvents.DestroyGrabLine:FireServer(charHRP) 
                        end
                        if SitMODED then 
                            charHUM.Sit = true 
                        end
                    end
                    
                    if isRidingNow then
                        rs.GrabEvents.SetNetworkOwner:FireServer(charHRP, CFrame.lookAt(myHRP.Position, charHRP.Position))
                        if not OnlyOwner then 
                            rs.GrabEvents.DestroyGrabLine:FireServer(charHRP) 
                        end
                        if SitMODED then  
                            charHUM.Sit = true 
                        end
                    end

                    if blobLoopT4 and bodyPositions[player] and bodyPositions[player].Parent then
                        -- 내 원래 위치 기준으로 타겟을 끌고오기
                        local targetPosition
                        if myOriginalPosition then
                            targetPosition = myOriginalPosition.Position + OLTPValue
                        else
                            targetPosition = myHRP.Position + OLTPValue
                        end
                        bodyPositions[player].Position = targetPosition
                    end
                    restoreMyPosition()
                end)

                if SkipOL then task.wait(2) end

                anchorToggleTasks[player] = task.spawn(function()
                    while blobLoopT4 and character and character.Parent and charHRP and charHRP.Parent and charHUM and charHUM.Health > 0 do
                        if charTOR and SkipOL then 
                            charTOR.Anchored = true 
                        end
                        task.wait()
                    end
                end)

                while blobLoopT4 do
                    if not character.Parent or not charHRP.Parent or charHUM.Health <= 0 then 
                        if independentHoldConnections[player] then
                            independentHoldConnections[player]:Disconnect()
                            independentHoldConnections[player] = nil
                        end
                        if anchorToggleTasks[player] then
                            task.cancel(anchorToggleTasks[player])
                            anchorToggleTasks[player] = nil
                        end
                        if ownerTags[player] then
                            ownerTags[player]:Destroy()
                            ownerTags[player] = nil
                        end
                        if charTOR then
                            charTOR.Anchored = false
                        end
                        cleanupBodyPosition(player)
                        break 
                    end

                    local currentSeat = plr.Character and plr.Character:FindFirstChildOfClass("Humanoid") and plr.Character:FindFirstChildOfClass("Humanoid").SeatPart
                    if currentSeat and currentSeat.Parent and currentSeat.Parent.Name == "CreatureBlobman" then
                        BlobGrab(currentBlobS, charHRP, "Right")
                        task.wait(0.03)
                        BlobRelease(currentBlobS, charHRP, "Right")
                        if GRABMODE then BlobGrab(currentBlobS, charHRP, "Right") end
                        charHUM.Sit = true
                        task.delay(0.02, function() if charHUM then charHUM.Sit = false end end)
                    end
                    task.wait(0.01)
                end

                if independentHoldConnections[player] then 
                    independentHoldConnections[player]:Disconnect()
                    independentHoldConnections[player] = nil
                end

                if anchorToggleTasks[player] then
                    task.cancel(anchorToggleTasks[player])
                    anchorToggleTasks[player] = nil
                end

                if ownerTags[player] then
                    ownerTags[player]:Destroy()
                    ownerTags[player] = nil
                end

                if charTOR then
                    charTOR.Anchored = false
                end

                cleanupBodyPosition(player)
            end

            if originPositions[player] then
                originPositions[player] = nil
            end
            
            task.wait(0.5)
        end
    end

    task.spawn(function()
        while blobLoopT4 do
            local targetList = {}
            if playersInLoop1V then for _, v in pairs(playersInLoop1V) do table.insert(targetList, v) end end
            if playersInLoop2V then for _, v in pairs(playersInLoop2V) do table.insert(targetList, v) end end

            for _, name in ipairs(targetList) do
                local player = game.Players:FindFirstChild(name)
                if not player or player == plr then continue end
                if PPs and (PPs:FindFirstChild(name) or (inv and inv:FindFirstChild(name))) then continue end

                if not monitoredPlayers[player] then

                            pcall(function() processPlayer(player) end)

                                independentHoldConnections[player]:Disconnect()
                                independentHoldConnections[player] = nil

                                anchorToggleTasks[player] = nil
                            end

                                ownerTags[player]:Destroy()
                                ownerTags[player] = nil
                            end
                            if originPositions[player] then
                                originPositions[player] = nil
                            end
                            cleanupBodyPosition(player)
                            lastTPTime[player] = nil
                            destroyCounts[player] = nil
                            if player.Character then
                                local charTOR = player.Character:FindFirstChild("Torso")
                                if charTOR then
                                    charTOR.Anchored = false
                                end
                            end
                        end)
                    end
                end

                    monitoredPlayers[player] = nil
                    if independentHoldConnections[player] then
                        independentHoldConnections[player]:Disconnect()
                        independentHoldConnections[player] = nil
                    end
                    if anchorToggleTasks[player] then
                        task.cancel(anchorToggleTasks[player])

                    end

                        ownerTags[player]:Destroy()

                    end

                        originPositions[player] = nil
                    end

                    lastTPTime[player] = nil
                    destroyCounts[player] = nil
                    if player.Character then
                        local charTOR = player.Character:FindFirstChild("Torso")
                        if charTOR then
                            charTOR.Anchored = false
                        end
                    end
                end
            end
            task.wait(0.5)
        end

        for player, task in pairs(anchorToggleTasks) do

            end
        end

        for _, taskInfo in pairs(runningTasks) do task.cancel(taskInfo) end
        for player, connection in pairs(independentHoldConnections) do
            if connection then connection:Disconnect() end

                end
            end
        end

        runningTasks = {}
        monitoredPlayers = {}
        independentHoldConnections = {}
        anchorToggleTasks = {}
        destroyCounts = {}
        bodyPositions = {}
        lastTPTime = {}
    end)
end

                    local ragdolled = humanoid:FindFirstChild("Ragdolled")

                        continue  -- 이미 래그돌 상태면 스킵
                    end
                end
                
                local otherHrp = other.Character:FindFirstChild("HumanoidRootPart")
                if not otherHrp then continue end
                
                -- 🔥 레그돌 효과 적용
                local movel = 0.1
                local CFCF = other.Character:GetPivot()
                
                -- 속도 저장 후 강제 이동
                local vel = otherHrp.Velocity
                otherHrp.Velocity = vel * 10000 + Vector3.new(0, 10000, 0)
                RunService.RenderStepped:Wait()
                otherHrp.Velocity = vel
                RunService.Stepped:Wait()
                otherHrp.Velocity = vel + Vector3.new(0, movel, 0)
                movel = -movel
                
                -- 위치 조작
                other.Character:PivotTo(CFrame.new(0, -20, 0))
                task.wait(0.05)
                other.Character:PivotTo(CFCF)
                
                break  -- 한 번에 한 명만 처리
            end
        end
        
        task.wait(0.1)
    end

-- 내 위치 저장
local function saveMyPosition()
    local char = plr.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp then
        myOriginalPosition = hrp.CFrame
    end
end

-- 내 위치 복귀
local function restoreMyPosition()
    local char = plr.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp and myOriginalPosition then
        hrp.CFrame = myOriginalPosition
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end
end

-- TP 함수 (상대방에게만)
local function TPtoTarget(target)
    local TCHAR = target.Character
    if not TCHAR then return false end
    local THRP = TCHAR:FindFirstChild("Torso") or TCHAR:FindFirstChild("HumanoidRootPart")
    if not THRP then return false end

    local localChar = plr.Character
    local localHRP = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not localHRP then return false end

    if TCHAR and THRP and localHRP then
        local ping = plr:GetNetworkPing()
        local offset = THRP.Position + (THRP.Velocity * (ping + 0.15))
        localHRP.CFrame = CFrame.new(offset) * THRP.CFrame.Rotation
        return true
    end
    return false
end

-- 현재 탄 블롭 찾기
local function getCurrentBlob()
    local char = plr.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end

    for _, blobs in workspace:GetDescendants() do
        if blobs.Name == "CreatureBlobman" then
            local seat = blobs:FindFirstChild("VehicleSeat")
            if seat then
                local weld = seat:FindFirstChild("SeatWeld")
                if weld and weld.Part1 == hrp then
                    return blobs
                end
            end
        end
    end
    return nil
end

-- 블롭 그랩
local function blobGrab(blob, target)
    if not blob or not target then return end
    local detector = blob:FindFirstChild("RightDetector")
    if not detector then return end
    local weld = detector:FindFirstChild("RightWeld")
    if not weld then return end
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript")
    if script and script:FindFirstChild("CreatureGrab") then
        pcall(function() script.CreatureGrab:FireServer(detector, target, weld) end)
    end
end

-- 블롭 릴리즈 (빠르게)
local function blobReleaseFast(blob, target)
    if not blob or not target then return end
    local detector = blob:FindFirstChild("RightDetector")
    if not detector then return end
    local weld = detector:FindFirstChild("RightWeld")
    if not weld then return end
    local script = blob:FindFirstChild("BlobmanSeatAndOwnerScript")
    if script and script:FindFirstChild("CreatureRelease") then
        pcall(function() script.CreatureRelease:FireServer(weld, target) end)
    end
end

-- 블롭 킬 루프 (제자리 복귀)
local function blobKillLoop()
    while blobKillActive do
        local myBlob = getCurrentBlob()
        if not myBlob then
            task.wait(0.3)
            continue
        end
        
        for _, targetName in ipairs(blobTargetList) do
            if not blobKillActive then break end
            
            local player = Players:FindFirstChild(targetName)
            if not player then continue end
            
            local char = player.Character
            if not char then continue end
            
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then continue end
            
            local hum = char:FindFirstChild("Humanoid")
            if not hum or hum.Health <= 0 then continue end
            
            -- 1. 내 위치 저장
            saveMyPosition()
            
            -- 2. TP로 접근
            TPtoTarget(player)
            task.wait(0.05)
            
            -- 3. 빠른 그랩 & 릴리즈
            for i = 1, 3 do
                if not blobKillActive then break end
                pcall(function() blobGrab(myBlob, hrp) end)
                task.wait(0.03)
                pcall(function() blobReleaseFast(myBlob, hrp) end)
                task.wait(0.03)
            end
            
            -- 4. 보이드로 보내기
            local FallenY = workspace.FallenPartsDestroyHeight
            local targetY = (FallenY <= -50000 and -49999) or (FallenY <= -100 and -99) or -100
            pcall(function() 
                hrp.CFrame = CFrame.new(99999, targetY, 99999)
                if hrp.AssemblyLinearVelocity then
                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                end
            end)
            
            task.wait(0.05)
            
            -- 5. 내 위치 복귀
            restoreMyPosition()
            
            task.wait(0.1)
        end
        
        task.wait(0.2)
    end
end

local SnowballRagdollEnabled = false
local SnowballRagdollThread = nil
local SnowballTargetIndex = 1

local function ExecuteSnowballRagdoll()
    while SnowballRagdollEnabled do
        if #setOwnerTargetList == 0 then
            task.wait(0.5)
            continue
        end
        
        if SnowballTargetIndex > #setOwnerTargetList then
            SnowballTargetIndex = 1
        end
        
        local targetName = setOwnerTargetList[SnowballTargetIndex]
        local target = Players:FindFirstChild(targetName)
        
        if not target or not target.Character then
            SnowballTargetIndex = SnowballTargetIndex + 1
            task.wait(0.1)
            continue
        end
        
        local myChar = plr.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local targetChar = target.Character
        local targetHrp = targetChar:FindFirstChild("HumanoidRootPart")
        
        if not myHrp or not targetHrp then
            SnowballTargetIndex = SnowballTargetIndex + 1
            task.wait(0.1)
            continue
        end
        
        local rSpawn = SpawnToyRemote
        local rOwner = SetNetworkOwner
        local rExplode = rs:FindFirstChild("BombEvents") and rs.BombEvents:FindFirstChild("BombExplode")
        
        if rSpawn and rOwner and rExplode then
            task.spawn(function()
                pcall(function()
                    rSpawn:InvokeServer("BallSnowball", myHrp.CFrame * CFrame.new(0, 10, 20), Vector3.new(0, 0, 0))
                end)
            end)
            
            task.wait(0.15)
            
            local inv = Workspace:FindFirstChild(plr.Name .. "SpawnedInToys")
            local ballPart = inv and inv:FindFirstChild("BallSnowball")
            local ballSPart = ballPart and ballPart:FindFirstChild("SoundPart")
            
            if ballPart and ballSPart then
                pcall(function()
                    rOwner:FireServer(ballSPart, ballSPart.CFrame)
                end)
                task.wait(0.05)
                
                ballSPart.CFrame = targetHrp.CFrame
                task.wait(0.05)
                
                pcall(function()
                    rExplode:FireServer({
                        Radius = 0,
                        Color = Color3.new(0, 0, 0),
                        TimeLength = 0,
                        Model = ballPart,
                        Type = "SnowPoof",
                        ExplodesByFire = false,
                        MaxForcePerStudSquared = 0,
                        Hitbox = ballSPart,
                        ImpactSpeed = 0,
                        ExplodesByPointy = false,
                        DestroysModel = true,
                        PositionPart = ballSPart
                    }, Vector3.new(0, 0, 0))
                end)
                
                if RagdollRemote and targetHrp then
                    pcall(function()
                        RagdollRemote:FireServer(targetHrp, 0)
                    end)
                end
            end
        end
        
        SnowballTargetIndex = SnowballTargetIndex + 1
        task.wait(0.15)
    end
end

local antiGrabConn = nil
local isvs = false
local savedPosition = nil  -- 저장된 원래 위치
local isUnderAttack = false  -- 공격 받는 중인지
local savePositionEnabled = true  -- 위치 저장 활성화 여부
local lastAttackTime = 0  -- 마지막 공격 시간
local attackCooldown = 0.5  -- 공격 쿨다운 (0.5초)

-- SetOwner 이벤트 감지용
local SetOwnerRemote = rs.GrabEvents.SetNetworkOwner
local DestroyGrabLineRemote = rs.GrabEvents.DestroyGrabLine

local function setRagdollF(state)
    local char = plr.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local RagdollRemote = rs and rs:FindFirstChild("CharacterEvents") and rs.CharacterEvents:FindFirstChild("RagdollRemote")
    if RagdollRemote then
        RagdollRemote:FireServer(hrp, state and 0 or 1)
    end
end

-- SetOwner/DestroyGrabLine 이벤트 감지 함수 (원본 유지)
local function setupEventDetection()
    -- SetOwner 이벤트 감지
    if SetOwnerRemote then
        local oldSetOwner = SetOwnerRemote.FireServer
        SetOwnerRemote.FireServer = function(self, target, ...)
            -- 내가 타겟인지 확인
            if target and target:IsA("BasePart") then
                local targetChar = target.Parent
                if targetChar == plr.Character then
                    lastAttackTime = tick()
                    if not isUnderAttack then
                        isUnderAttack = true
                        savePositionEnabled = false
                        if savedPosition then
                            local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
                            if hrp then
                                hrp.CFrame = savedPosition
                                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                            end
                        end
                    end
                end
            end
            return oldSetOwner(self, target, ...)
        end
    end
    
    -- DestroyGrabLine 이벤트 감지
    if DestroyGrabLineRemote then
        local oldDestroyGrabLine = DestroyGrabLineRemote.FireServer
        DestroyGrabLineRemote.FireServer = function(self, target, ...)
            -- 내가 타겟인지 확인
            if target and target:IsA("BasePart") then
                local targetChar = target.Parent
                if targetChar == plr.Character then
                    lastAttackTime = tick()
                    if not isUnderAttack then
                        isUnderAttack = true
                        savePositionEnabled = false
                        if savedPosition then
                            local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
                            if hrp then
                                hrp.CFrame = savedPosition
                                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                            end
                        end
                    end
                end
            end
            return oldDestroyGrabLine(self, target, ...)
        end
    end
end

-- 이벤트 감지 설정 실행
task.spawn(setupEventDetection)

local function AntiGrabF(enable)
    if antiGrabConn then
        antiGrabConn:Disconnect()
        antiGrabConn = nil
    end

    if not enable then 
        local char = plr.Character
        if char then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChild("Humanoid")
            if hum then hum.RequiresNeck = true end
            if hrp and hrp.Anchored then hrp.Anchored = false end
            if hum then
                hum.Sit = false
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end
        savedPosition = nil
        isUnderAttack = false
        savePositionEnabled = true
        return 
    end

    -- 초기 위치 저장
    local function saveInitialPosition()
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp and not savedPosition then
            savedPosition = hrp.CFrame
        end
    end
    saveInitialPosition()

    -- 위치 저장 스레드 (공격 받지 않을 때만 저장)
    task.spawn(function()
        while enable do
            if savePositionEnabled and not isUnderAttack then
                local char = plr.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    savedPosition = hrp.CFrame
                end
            end
            task.wait(0.1)  -- 0.1초마다 저장
        end
    end)

    -- 공격 상태 체크 스레드 (일정 시간 공격 없으면 상태 해제)
    task.spawn(function()
        while enable do
            if isUnderAttack then
                local now = tick()
                if now - lastAttackTime > attackCooldown then
                    -- 0.5초 동안 공격 없으면 상태 해제
                    isUnderAttack = false
                    savePositionEnabled = true
                end
            end
            task.wait(0.1)
        end
    end)

    local lastHeldState = false
    local sitHoldTimer = 0
    local shouldKeepSit = false
    local wasHeld = false
    local lastRagdollTime = 0
    local ragdollDuration = 0.48

    antiGrabConn = RunService.Heartbeat:Connect(function(deltaTime)
        local char = plr.Character
        if not char then return end
        local hum = char:FindFirstChild("Humanoid")
        if not hum then return end
        
        isvs = hum and hum.SeatPart ~= nil
        local isHeld = plr:FindFirstChild("IsHeld")
        if not isHeld then return end

        local head = char:FindFirstChild("Head")
        local POR = head and head:FindFirstChild("PartOwner")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hrp2 = char:FindFirstChild("Torso")
        
        if not hrp or not hum then return end

        local FPDH = workspace.FallenPartsDestroyHeight
        local DY = (FPDH <= -50000 and -49999) or (FPDH <= -100 and -99) or -100
        local now = tick()

        if hum then
            hum.RequiresNeck = false
            hum.AutoRotate = true
        end

        if isvs and POR then task.wait(0.3) end
        if isvs then task.wait(0.3) end

        local rag = hum:FindFirstChild("Ragdolled")
        
        -- ===== IsHeld 또는 PartOwner 감지 시 복귀 =====
        if isHeld.Value == true or POR then
            if not isUnderAttack then
                isUnderAttack = true
                savePositionEnabled = false
                lastAttackTime = now
            end
            
            -- 저장된 위치로 복귀
            if savedPosition then
                hrp.CFrame = savedPosition
            end
            
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            
            if hrp2 then
                hrp2.CFrame = hrp.CFrame
            end
        end
        -- ===== 끝 =====

        if isHeld.Value == true and rag and rag.Value == true then
            for _, limbName in ipairs({"Head", "Left Arm", "Right Arm", "Left Leg", "Right Leg"}) do
                local limb = char:FindFirstChild(limbName)
                if limb then
                    local ragdollPart = limb:FindFirstChild("RagdollLimbPart")
                    local ragdollCons = limb:FindFirstChild("BallSocketConstraint")
                    if ragdollCons then ragdollCons.Enabled = false end
                    if ragdollPart then ragdollPart.CanCollide = false end
                end
            end
        end

        if hum.Health <= 0 then
            lastHeldState = false
            shouldKeepSit = false
            sitHoldTimer = 0
            wasHeld = false
            hum.Sit = false
            hum.AutoRotate = true
            isUnderAttack = false
            savePositionEnabled = true
        end

        if not hrp or hum.Health <= 0 then
            local Struggle = rs and rs:FindFirstChild("CharacterEvents") and rs.CharacterEvents:FindFirstChild("Struggle")
            if Struggle then Struggle:FireServer() end
            if hrp2 then hrp2.CFrame = CFrame.new(9999, DY, 9999) end
        end

        if isHeld.Value == true then
            if hum.MoveDirection.Magnitude > 0 then
                local moveSpeed = 10
                local moveVector = hum.MoveDirection * deltaTime * moveSpeed
                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                moveVector = Vector3.new(moveVector.X, 0, moveVector.Z)
                hrp.CFrame = hrp.CFrame + moveVector
                if hrp2 then hrp2.CFrame = hrp2.CFrame + moveVector end
            end
        end

        if isHeld.Value ~= lastHeldState then
            if isHeld.Value == true then
                wasHeld = true
                shouldKeepSit = true
                sitHoldTimer = 0.3
                hum:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
                hum.Sit = true
                hum.AutoRotate = true
                lastRagdollTime = now
                if POR and hrp then
                    local Struggle = rs and rs:FindFirstChild("CharacterEvents") and rs.CharacterEvents:FindFirstChild("Struggle")
                    local RagdollRemote = rs and rs:FindFirstChild("CharacterEvents") and rs.CharacterEvents:FindFirstChild("RagdollRemote")
                    if Struggle then Struggle:FireServer() end
                    if RagdollRemote then RagdollRemote:FireServer(hrp, 0) end
                end
                if hrp then setRagdollF(true) end
            else
                if wasHeld then
                    shouldKeepSit = true
                    sitHoldTimer = 0.3
                end
            end
            lastHeldState = isHeld.Value
        end

        if lastRagdollTime > 0 and now - lastRagdollTime >= ragdollDuration then
            setRagdollF(false)
            lastRagdollTime = 0
        end

        if isHeld.Value == true and lastRagdollTime > 0 and now - lastRagdollTime < ragdollDuration then
            if hrp then setRagdollF(true) end
        end

        if sitHoldTimer > 0 then
            sitHoldTimer = sitHoldTimer - deltaTime
            if isHeld.Value == true or POR or (rag and rag.Value == true) then
                shouldKeepSit = true
                sitHoldTimer = 0.3
                hum.Sit = true
                if isHeld.Value == true then
                    hum:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
                end
            else
                hum.Sit = true
            end
        end

        if sitHoldTimer <= 0 and shouldKeepSit then
            local currentRagdolled = hum:FindFirstChild("Ragdolled")
            if not currentRagdolled or currentRagdolled.Value == false then
                hum.Sit = false
                hum:ChangeState(Enum.HumanoidStateType.Running)
                shouldKeepSit = false
                wasHeld = false
            else
                sitHoldTimer = 0.3
            end
        end

        if shouldKeepSit then hum.Sit = true end

        if POR then
            local attackerName = POR.Value
            local Struggle = rs and rs:FindFirstChild("CharacterEvents") and rs.CharacterEvents:FindFirstChild("Struggle")
            local RagdollRemote = rs and rs:FindFirstChild("CharacterEvents") and rs.CharacterEvents:FindFirstChild("RagdollRemote")
            if Struggle then Struggle:FireServer() end
            if RagdollRemote then RagdollRemote:FireServer(hrp, 0) end
            shouldKeepSit = true
            sitHoldTimer = 0.3
            hum.Sit = true
            hum.AutoRotate = true
            lastRagdollTime = now
            if hrp then setRagdollF(true) end
        end
    end)
end

-- 안티렉 변수
local antiLagEnabled = false
local antiLagThreshold = 10  -- 초당 허용 횟수
local fireCount = {}

-- 안티렉 함수 (message 26 기반)
local function setupAntiLag()
    local Pline = rs.GrabEvents.CreateGrabLine.OnClientEvent
    
    Pline:Connect(function(fromPlr)
        if not antiLagEnabled then return end
        if typeof(fromPlr) ~= "Instance" or not fromPlr:IsA("Player") then return end
        if fromPlr == plr then return end

        local now = os.clock()

        if not fireCount[fromPlr] then 
            fireCount[fromPlr] = {count = 0, start = now} 
        end

        local data = fireCount[fromPlr]

        if now - data.start > 1 then
            data.count = 0
            data.start = now
        end

        data.count = data.count + 1

        if data.count >= antiLagThreshold and not data.isDecreasing then
            data.isDecreasing = true

            -- 로컬 플레이어의 빔 렌더링 스크립트 비활성화 (렉 방지)
            local beamScript = plr.PlayerScripts:FindFirstChild("CharacterAndBeamMove")
            if beamScript then 
                beamScript.Enabled = false 
            end

            task.spawn(function()
                while data.count > 0 do
                    task.wait(0.1)
                    data.count = data.count - 1
                end
                if beamScript then 
                    beamScript.Enabled = true 
                end
                fireCount[fromPlr] = nil
            end)
        end
    end)
end

-- 변수
local AutoAttackerEnabled = false
local AutoAttackerThread = nil
local trackedAttackers = {}

-- =============================================
-- [ 공격자 감지 및 처치 함수 ]
-- =============================================
local function detectAndPunish()
    local invName = plr.Name .. "SpawnedInToys"
    
    while AutoAttackerEnabled do
        local myChar = plr.Character
        local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local myAttach = myHRP and myHRP:FindFirstChild("RootAttachment")
        
        if myChar and myHRP and myAttach then
            -- 1. 내 머리에 PartOwner가 있으면 (누군가 나를 잡고 있음)
            local head = myChar:FindFirstChild("Head")
            if head then
                local ownerVal = head:FindFirstChild("PartOwner")
                if ownerVal and ownerVal:IsA("StringValue") and ownerVal.Value ~= "" then
                    local attacker = Players:FindFirstChild(ownerVal.Value)
                    if attacker and attacker ~= plr then
                        trackedAttackers[attacker.Name] = attacker
                    end
                end
            end
            
            -- 2. 내 블롭이 나를 잡고 있는지 확인 (다른 사람의 블롭)
            local inv = Workspace:FindFirstChild(invName)
            if inv then
                for _, blob in ipairs(inv:GetChildren()) do
                    if blob.Name == "CreatureBlobman" and blob:FindFirstChild("BlobmanSeatAndOwnerScript") then
                        for _, side in ipairs({"Left", "Right"}) do
                            local detector = blob:FindFirstChild(side.."Detector")
                            local weld = detector and detector:FindFirstChild(side.."Weld")
                            if weld and weld:IsA("AlignPosition") and weld.Attachment0 == myAttach then
                                -- 내 블롭에 내가 잡혀있음 → 블롭 주인이 공격자
                                local blobOwner = nil
                                for _, player in ipairs(Players:GetPlayers()) do
                                    if player.Character and blob:IsDescendantOf(Workspace:FindFirstChild(player.Name.."SpawnedInToys")) then
                                        blobOwner = player
                                        break
                                    end
                                end
                                if blobOwner and blobOwner ~= plr then
                                    trackedAttackers[blobOwner.Name] = blobOwner
                                end
                            end
                        end
                    end
                end
            end
            
            -- 3. 다른 사람의 블롭에 내가 잡혀있는지 확인
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= plr then
                    local invs = Workspace:FindFirstChild(player.Name.."SpawnedInToys")
                    if invs then
                        for _, blob in ipairs(invs:GetChildren()) do
                            if blob.Name == "CreatureBlobman" and blob:FindFirstChild("BlobmanSeatAndOwnerScript") then
                                for _, side in ipairs({"Left", "Right"}) do
                                    local detector = blob:FindFirstChild(side.."Detector")
                                    local weld = detector and detector:FindFirstChild(side.."Weld")
                                    if weld and weld:IsA("AlignPosition") and weld.Attachment0 == myAttach then
                                        trackedAttackers[player.Name] = player
                                    end
                                end
                            end
                        end
                    end
                end
            end
            
            -- 4. 플롯 아이템에 있는 블롭 확인
            local plots = Workspace:FindFirstChild("PlotItems")
            if plots then
                for i = 1, 5 do
                    local plot = plots:FindFirstChild("Plot"..i)
                    if plot then
                        for _, blob in ipairs(plot:GetChildren()) do
                            if blob.Name == "CreatureBlobman" and blob:FindFirstChild("BlobmanSeatAndOwnerScript") then
                                for _, side in ipairs({"Left", "Right"}) do
                                    local detector = blob:FindFirstChild(side.."Detector")
                                    local weld = detector and detector:FindFirstChild(side.."Weld")
                                    if weld and weld:IsA("AlignPosition") and weld.Attachment0 == myAttach then
                                        local plotOwner = nil
                                        local plotSign = plot:FindFirstChild("PlotSign")
                                        if plotSign and plotSign:FindFirstChild("ThisPlotsOwners") then
                                            for _, owner in ipairs(plotSign.ThisPlotsOwners:GetChildren()) do
                                                if owner:IsA("StringValue") then
                                                    plotOwner = Players:FindFirstChild(owner.Value)
                                                    if plotOwner then break end
                                                end
                                            end
                                        end
                                        if plotOwner and plotOwner ~= plr then
                                            trackedAttackers[plotOwner.Name] = plotOwner
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
        
        -- =============================================
        -- [ 감지된 공격자 처치 ]
        -- =============================================
        for attackerName, attacker in pairs(trackedAttackers) do
            if attacker and attacker.Character then
                local targetChar = attacker.Character
                local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
                local targetHead = targetChar:FindFirstChild("Head")
                local targetTorso = targetChar:FindFirstChild("Torso")
                local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
                
                if targetHRP and targetHum and targetHum.Health > 0 then
                    -- SetNetworkOwner로 권한 획득
                    pcall(function()
                        if SetNetworkOwner then
                            SetNetworkOwner:FireServer(targetHRP, targetHRP.CFrame)
                            if targetTorso then
                                SetNetworkOwner:FireServer(targetTorso, targetTorso.CFrame)
                            end
                        end
                    end)
                    
                    task.wait(0.05)
                    
                    -- 보이드로 보내기 (죽이기)
                    local FallenY = Workspace.FallenPartsDestroyHeight
                    local targetY = (FallenY <= -50000 and -49999) or (FallenY <= -100 and -99) or -100
                    
                    pcall(function()
                        targetHRP.CFrame = CFrame.new(99999, targetY, 99999)
                        if targetTorso then
                            targetTorso.CFrame = CFrame.new(99999, targetY, 99999)
                        end
                    end)                    
                    
                    -- 추적 목록에서 제거 (한 번 처치하면 다시 감지될 때까지 대기)
                    trackedAttackers[attackerName] = nil
                end
            else
                -- 캐릭터가 없으면 목록에서 제거
                trackedAttackers[attackerName] = nil
            end
        end
        
        task.wait(0.1) -- 감지 주기
    end
end

-- 상대방 블롭만 제거하는 함수
local function KillEnemyBlobs()
    if not EnemyBlobKillEnabled then return end
    
    local SetNetworkOwner = rs and rs:FindFirstChild("GrabEvents") and rs.GrabEvents:FindFirstChild("SetNetworkOwner")
    local DestroyToy = rs and rs:FindFirstChild("MenuToys") and rs.MenuToys:FindFirstChild("DestroyToy")
    
    -- 내가 생성한 블롭 폴더
    local myInv = workspace:FindFirstChild(plr.Name .. "SpawnedInToys")
    
    -- 모든 블롭 순회
    for _, blob in ipairs(workspace:GetDescendants()) do
        if blob.Name == "CreatureBlobman" then
            -- 내가 생성한 블롭이면 스킵 (죽이지 않음)
            if myInv and blob:IsDescendantOf(myInv) then
                continue -- 내 블롭은 보호
            end
            
            -- 상대방 블롭만 제거
            local blobHumanoid = blob:FindFirstChild("HumanoidCreature") or blob:FindFirstChildOfClass("Humanoid")
            local blobHRP = blob:FindFirstChild("HumanoidRootPart")
            
            -- 소유권 획득 시도
            if SetNetworkOwner and blobHRP then
                pcall(function()
                    SetNetworkOwner:FireServer(blobHRP, blobHRP.CFrame)
                end)
            end
            
            -- 체력 0으로 만들기
            if blobHumanoid then
                pcall(function()
                    blobHumanoid.Health = 0
                end)
            end
            
            -- 블롭 제거
            if DestroyToy then
                pcall(function()
                    DestroyToy:FireServer(blob)
                end)
            end
        end
    end
end

local PencilCount = 0

local function GetMyPlotNumber()
    local char = plr.Character
    if not char then return nil end
    
    if char.Parent and char.Parent.Name == "PlayersInPlots" then
        for _, plot in workspace.Plots:GetChildren() do
            local plotSign = plot:FindFirstChild("PlotSign")
            if plotSign and plotSign:FindFirstChild("ThisPlotsOwners") then
                for _, owner in plotSign.ThisPlotsOwners:GetChildren() do
                    if owner.Value == plr.Name then
                        if plot.Name == "Plot1" then
                            return 1
                        elseif plot.Name == "Plot2" then
                            return 2
                        elseif plot.Name == "Plot3" then
                            return 3
                        elseif plot.Name == "Plot4" then
                            return 4
                        elseif plot.Name == "Plot5" then
                            return 5
                        end
                    end
                end
            end
        end
    end
    return nil
end

local function GetInventory()
    local plotNumber = GetMyPlotNumber()
    
    if plotNumber then
        local plotItems = workspace:FindFirstChild("PlotItems")
        if plotItems then
            local plotFolder = plotItems:FindFirstChild("Plot" .. plotNumber)
            if plotFolder then
                return plotFolder
            end
        end
    end
    
    return workspace:FindFirstChild(plr.Name .. "SpawnedInToys") or workspace:FindFirstChild("SpawnedInToys")
end

local function createPencilChain(count)
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    
    if not count or count <= 0 or count > 50 then
        Rayfield:Notify({
            Title = "❌ 오류",
            Content = "1~50 사이의 숫자를 입력하세요",
            Duration = 2
        })
        return
    end
    
    local character = plr.Character or plr.CharacterAdded:Wait()
    local hrp = character:WaitForChild("HumanoidRootPart")
    local torso = character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
    
    if not torso then
        Rayfield:Notify({ Title = "❌ 오류", Content = "캐릭터를 찾을 수 없음", Duration = 2 })
        return
    end
    
    local StickyPartEvent = ReplicatedStorage:WaitForChild("PlayerEvents"):WaitForChild("StickyPartEvent")
    local SpawnToy = ReplicatedStorage:WaitForChild("MenuToys"):WaitForChild("SpawnToyRemoteFunction")
    
    local function createPencil(index)
        while not plr.CanSpawnToy.Value do task.wait() end
        
        task.spawn(function()
            pcall(function()
                SpawnToy:InvokeServer("ToolPencil", hrp.CFrame * CFrame.new(0, 5, 15), Vector3.new(0, 0, 0))
            end)
        end)
        
        task.wait(0.3)
        
        local inv = GetInventory()
        if not inv then
            return nil
        end
        
        local pencil = nil
        local children = inv:GetChildren()
        for i = #children, 1, -1 do
            if children[i].Name == "ToolPencil" then
                pencil = children[i]
                break
            end
        end
        
        if not pencil then 
            return nil 
        end
        
        local stickyPart = pencil:WaitForChild("StickyPart")
        local SoundPart = pencil:WaitForChild("SoundPart")
        
        if stickyPart and SoundPart then
            pcall(function()
                ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(SoundPart, SoundPart.CFrame)
                ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(SoundPart, torso.CFrame)
            end)
        end
        
        local PartOwner = SoundPart:FindFirstChild("PartOwner")
        if PartOwner then
            SoundPart.CFrame = CFrame.new(0, 599999, 999)
        end
        
        stickyPart.Name = "Pencil_" .. index
        
        for _, part in pairs(pencil:GetDescendants()) do
            if part:IsA("BasePart") and part.Color == Color3.fromRGB(158, 108, 141) then
                part.Name = "Connector_" .. index
            end
        end
        
        return pencil
    end
    
    local pencils = {}
    for i = 1, count do
        local pencil = createPencil(i)
        if pencil then
            table.insert(pencils, pencil)
        end
        task.wait(0.1)
    end
    
    if #pencils == 0 then
        Rayfield:Notify({ Title = "❌ 오류", Content = "연필 생성 실패", Duration = 2 })
        return
    end
    
    for i = 1, #pencils - 1 do
        local currentPencil = pencils[i]
        local nextPencil = pencils[i + 1]
        
        if currentPencil and nextPencil then
            local attachFrom = currentPencil:FindFirstChild("Pencil_" .. i)
            local attachTo = nextPencil:FindFirstChild("Connector_" .. (i + 1))
            
            if attachFrom and attachTo then
                pcall(function()
                    StickyPartEvent:FireServer(attachFrom, attachTo, CFrame.Angles(0, math.rad(-90), 0))
                end)
            end
        end
        task.wait(0.05)
    end
    
    local hrp = character:FindFirstChild("HumanoidRootPart")
    local torso2 = hrp and hrp:FindFirstChild("RagdollTouchedHitbox")
    
    if torso2 and #pencils >= count then
        local lastPencil = pencils[count]
        if lastPencil then
            local lastAttach = lastPencil:FindFirstChild("Pencil_" .. count)
            if lastAttach then
                local attachPosition = torso2.CFrame * CFrame.new(0, -1.1, 0.1) * CFrame.Angles(0, math.rad(180), 0)
                local relativeCF = torso2.CFrame:ToObjectSpace(attachPosition)
                pcall(function()
                    StickyPartEvent:FireServer(lastAttach, torso2, relativeCF)
                end)
            end
        end
    end
    
    Rayfield:Notify({
        Title = "✅ 연필 연결 완료",
        Content = count .. "개 연결됨",
        Duration = 2
    })
end

OtherTab:CreateInput({
    Name = "연필 개수",
    CurrentValue = "5",
    PlaceholderText = "1~50",
    RemoveTextAfterFocusLost = false,
    Callback = function(Value)
        local num = tonumber(Value)
        if num and num > 0 and num <= 50 then
            PencilCount = num
        end
    end
})

OtherTab:CreateButton({
    Name = "✏️ 연필 연결 생성",
    Callback = function()
        if PencilCount <= 0 then
            Rayfield:Notify({ Title = "❌ 오류", Content = "개수를 입력하세요", Duration = 2 })
            return
        end
        createPencilChain(PencilCount)
    end
})
